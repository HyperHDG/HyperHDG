set(HYPERHDG_USE_DEFINITIONS)
set(HYPERHDG_USE_LIBS)
function(add_subdirectory_if DIR)
  option(HYPERHDG_USE_${DIR} "use submodule '${DIR}'" ON)
  if (HYPERHDG_USE_${DIR})
    message("HyperHDG: using ${DIR}")
    add_subdirectory(${DIR})
    list(APPEND HYPERHDG_USE_DEFINITIONS -DHYPERHDG_USE_${DIR})
    list(APPEND HYPERHDG_USE_LIBRARIES ${DIR})
  else()
    message("HyperHDG: not using ${DIR}")
  endif()
endfunction()

option(HYPERHDG_USE_PARHIP "use parhip submodule of kahip" OFF)

else()

endif()
